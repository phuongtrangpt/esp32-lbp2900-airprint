cmake_minimum_required(VERSION 3.16.0)
# explicit version: ESP-IDF otherwise runs `git describe`, which fails in a
# repository without commits
set(PROJECT_VER "0.9.12-wifi2")
set(PRINTSERVER_PROFILE "all" CACHE STRING "Printer profile: hp, hp_p1008, canon, pixma, standard, all")
if(PRINTSERVER_PROFILE STREQUAL "hp_pc")
  add_compile_definitions(PRINTSERVER_PROFILE_HP=1 PRINTSERVER_HP_RAW_BRIDGE=1 PRINTSERVER_HP_PC_OFFLOAD=1)
elseif(PRINTSERVER_PROFILE STREQUAL "hp")
  add_compile_definitions(PRINTSERVER_PROFILE_HP=1)
elseif(PRINTSERVER_PROFILE STREQUAL "hp_p1008")
  add_compile_definitions(PRINTSERVER_PROFILE_HP=1 PRINTSERVER_PROFILE_HP_P1008=1)
elseif(PRINTSERVER_PROFILE STREQUAL "canon")
  add_compile_definitions(PRINTSERVER_PROFILE_CANON=1)
elseif(PRINTSERVER_PROFILE STREQUAL "pixma")
  add_compile_definitions(PRINTSERVER_PROFILE_PIXMA=1)
elseif(PRINTSERVER_PROFILE STREQUAL "standard")
  add_compile_definitions(PRINTSERVER_PROFILE_STANDARD=1)
elseif(NOT PRINTSERVER_PROFILE STREQUAL "all")
  message(FATAL_ERROR "Unknown PRINTSERVER_PROFILE")
endif()
if(PRINTSERVER_PROFILE STREQUAL "hp")
  if(PRINTSERVER_HP_RAW_BRIDGE)
    set(PROJECT_VER "0.9.10-hp-raw12-net3")
    add_compile_definitions(PRINTSERVER_HP_RAW_BRIDGE=1)
  else()
    set(PROJECT_VER "0.9.10-hp-test4")
  endif()
endif()
if(PRINTSERVER_PROFILE STREQUAL "hp_pc")
  set(PROJECT_VER "0.1.1-hp-pc-usbwait")
endif()
message(STATUS "Printer profile: ${PRINTSERVER_PROFILE}")
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
if(PRINTSERVER_PROFILE STREQUAL "hp_pc")
  project(vietphuong_hp_pc)
elseif(PRINTSERVER_PROFILE STREQUAL "hp")
  project(vietphuong_hp)
elseif(PRINTSERVER_PROFILE STREQUAL "hp_p1008")
  project(vietphuong_hp_p1008)
elseif(PRINTSERVER_PROFILE STREQUAL "canon")
  project(vietphuong_canon)
elseif(PRINTSERVER_PROFILE STREQUAL "pixma")
  project(vietphuong_pixma)
elseif(PRINTSERVER_PROFILE STREQUAL "standard")
  project(vietphuong_standard)
else()
  project(usb_host_probe)
endif()
